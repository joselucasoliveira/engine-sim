cmake_minimum_required(VERSION 3.25)
project(EngineSimFleetNativeBench LANGUAGES CXX)
find_package(EngineSimFleet 1.0 CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(fleet-paced bench.cpp)
target_compile_features(fleet-paced PRIVATE cxx_std_17)
target_link_libraries(fleet-paced PRIVATE EngineSim::Fleet Threads::Threads)
option(ENGINE_SIM_FLEET_BENCH_PORTAUDIO "Build the Linux ALSA physical benchmark host" OFF)
if (ENGINE_SIM_FLEET_BENCH_PORTAUDIO)
    if (NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The physical benchmark host requires Linux ALSA")
    endif ()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PortAudio REQUIRED IMPORTED_TARGET portaudio-2.0>=19)
    target_compile_definitions(fleet-paced PRIVATE ENGINE_SIM_FLEET_BENCH_PORTAUDIO)
    target_link_libraries(fleet-paced PRIVATE PkgConfig::PortAudio)
endif ()
add_executable(fleet-stem-replay replay.cpp)
set_target_properties(fleet-stem-replay PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_link_libraries(fleet-stem-replay PRIVATE EngineSim::Fleet Threads::Threads)

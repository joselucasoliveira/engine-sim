cmake_minimum_required(VERSION 3.25)
project(RuntimeInstalledMatrix LANGUAGES C CXX)
find_package(EngineSimRuntime 2.0 CONFIG REQUIRED)
add_executable(runtime-matrix-c matrix.c)
set_target_properties(runtime-matrix-c PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO LINKER_LANGUAGE CXX)
add_executable(runtime-matrix-cxx matrix.cpp)
set_target_properties(runtime-matrix-cxx PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
foreach(target IN ITEMS runtime-matrix-c runtime-matrix-cxx)
    target_link_libraries(${target} PRIVATE EngineSim::Runtime)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic-errors)
    endif()
endforeach()

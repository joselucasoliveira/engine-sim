cmake_minimum_required(VERSION 3.25)

if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(EngineSimFleetExamples LANGUAGES C CXX)
    find_package(EngineSimFleet 1.0 CONFIG REQUIRED)
endif ()

add_executable(engine-sim-fleet-c-host host.c)
set_target_properties(engine-sim-fleet-c-host PROPERTIES
    C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO LINKER_LANGUAGE CXX)
add_executable(engine-sim-fleet-cxx-host host.cpp)
set_target_properties(engine-sim-fleet-cxx-host PROPERTIES
    CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)

foreach(target IN ITEMS engine-sim-fleet-c-host engine-sim-fleet-cxx-host)
    target_link_libraries(${target} PRIVATE EngineSim::Fleet)
    if (COMMAND engine_sim_diagnose)
        engine_sim_diagnose(${target} TRUE)
    endif ()
    if (MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else ()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic-errors)
    endif ()
endforeach ()

if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    option(ENGINE_SIM_FLEET_PORTAUDIO "Build the separate live host with a system PortAudio v19 shared library" OFF)
    if (ENGINE_SIM_FLEET_PORTAUDIO)
        find_package(portaudio 19 CONFIG QUIET)
        if (TARGET portaudio)
            get_target_property(pa_kind portaudio TYPE)
            if (NOT pa_kind STREQUAL "SHARED_LIBRARY")
                message(FATAL_ERROR "Select the upstream PortAudio v19 shared target")
            endif ()
            set(pa_target portaudio)
        elseif (UNIX)
            find_package(PkgConfig REQUIRED)
            pkg_check_modules(PortAudio REQUIRED IMPORTED_TARGET portaudio-2.0)
            set(saved_library_suffixes "${CMAKE_FIND_LIBRARY_SUFFIXES}")
            if (APPLE)
                set(CMAKE_FIND_LIBRARY_SUFFIXES .dylib)
            else ()
                set(CMAKE_FIND_LIBRARY_SUFFIXES .so)
            endif ()
            find_library(pa_shared NAMES portaudio HINTS ${PortAudio_LIBRARY_DIRS} REQUIRED)
            set(CMAKE_FIND_LIBRARY_SUFFIXES "${saved_library_suffixes}")
            unset(saved_library_suffixes)
            add_library(fleet_portaudio_provider SHARED IMPORTED)
            set_target_properties(fleet_portaudio_provider PROPERTIES IMPORTED_LOCATION "${pa_shared}"
                INTERFACE_INCLUDE_DIRECTORIES "${PortAudio_INCLUDE_DIRS}" INTERFACE_COMPILE_OPTIONS "${PortAudio_CFLAGS_OTHER}")
            set(pa_target fleet_portaudio_provider)
        else ()
            message(FATAL_ERROR "Provide a configured upstream PortAudio v19 shared package")
        endif ()
        add_executable(engine-sim-fleet-live live.cpp)
        target_compile_features(engine-sim-fleet-live PRIVATE cxx_std_17)
        set_target_properties(engine-sim-fleet-live PROPERTIES CXX_EXTENSIONS NO)
        target_link_libraries(engine-sim-fleet-live PRIVATE EngineSim::Fleet ${pa_target} "${CMAKE_DL_LIBS}")
    endif ()
endif ()

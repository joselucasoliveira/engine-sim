cmake_minimum_required(VERSION 3.25)
project(EngineSimProgrammatic LANGUAGES CXX)

option(ENGINE_SIM_PROGRAMMATIC_RUNTIME "Build the runtime byte and file consumers" ON)
find_package(EngineSimConfig 1.0 CONFIG REQUIRED COMPONENTS Json)
add_executable(engine-sim-model model.cpp)
target_link_libraries(engine-sim-model PRIVATE EngineSim::Json)
set_target_properties(engine-sim-model PROPERTIES CXX_EXTENSIONS NO)

if (ENGINE_SIM_PROGRAMMATIC_RUNTIME)
    enable_language(C)
    find_package(EngineSimRuntime 2.0 CONFIG REQUIRED)
    add_executable(engine-sim-programmatic programmatic.cpp)
    target_link_libraries(engine-sim-programmatic PRIVATE EngineSim::Json EngineSim::Runtime)
    set_target_properties(engine-sim-programmatic PROPERTIES CXX_EXTENSIONS NO)
    add_executable(engine-sim-file-loader file_loader.c)
    target_link_libraries(engine-sim-file-loader PRIVATE EngineSim::Runtime)
    set_target_properties(engine-sim-file-loader PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO LINKER_LANGUAGE CXX)
endif ()
if (MSVC)
    foreach(target IN ITEMS engine-sim-model engine-sim-programmatic engine-sim-file-loader)
        if (TARGET ${target})
            set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
        endif ()
    endforeach ()
endif ()

cmake_minimum_required(VERSION 3.25)
if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(EngineSimRuntimeExamples LANGUAGES C CXX)
    find_package(EngineSimRuntime 1.0 CONFIG REQUIRED)
    include(GNUInstallDirs)
else ()
    enable_language(C)
endif ()
add_executable(engine-sim-render render.cpp)
target_link_libraries(engine-sim-render PRIVATE EngineSim::Runtime)
target_compile_features(engine-sim-render PRIVATE cxx_std_17)
add_executable(engine-sim-runtime-consumer consumer/consumer.c)
target_link_libraries(engine-sim-runtime-consumer PRIVATE EngineSim::Runtime)
set_target_properties(engine-sim-runtime-consumer PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO LINKER_LANGUAGE CXX)
foreach(target IN ITEMS engine-sim-render engine-sim-runtime-consumer)
    if (COMMAND engine_sim_diagnose)
        engine_sim_diagnose(${target} TRUE)
    endif ()
    if (MSVC)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    elseif (APPLE)
        set_property(TARGET ${target} PROPERTY INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
    else ()
        set_property(TARGET ${target} PROPERTY INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
    endif ()
    install(TARGETS ${target} RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}" COMPONENT RuntimeSDK)
endforeach ()

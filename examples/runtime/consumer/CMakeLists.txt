cmake_minimum_required(VERSION 3.25)
project(EngineSimConsumer LANGUAGES C CXX)
find_package(EngineSimRuntime 2.0 CONFIG REQUIRED)
add_executable(runtime-c-consumer consumer.c)
set_target_properties(runtime-c-consumer PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO LINKER_LANGUAGE CXX)
add_executable(runtime-cxx-consumer consumer.cpp)
set_target_properties(runtime-cxx-consumer PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
foreach(target IN ITEMS runtime-c-consumer runtime-cxx-consumer)
    target_link_libraries(${target} PRIVATE EngineSim::Runtime)
    if (MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    else ()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic-errors)
    endif ()
endforeach ()
# Optional inspection for independent packaging tests; ordinary consumers need none.
if (ENGINE_SIM_CONSUMER_INSPECT)
    get_property(imported DIRECTORY PROPERTY IMPORTED_TARGETS)
    set(report "")
    foreach(target IN LISTS imported)
        if (target MATCHES "^EngineSim")
            foreach(property IN ITEMS TYPE IMPORTED_LOCATION IMPORTED_IMPLIB INTERFACE_INCLUDE_DIRECTORIES INTERFACE_LINK_LIBRARIES INTERFACE_COMPILE_DEFINITIONS)
                get_target_property(value ${target} ${property})
                string(APPEND report "${target}.${property}=${value}\n")
            endforeach ()
        endif ()
    endforeach ()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/imported-properties.txt" "${report}")
endif ()
